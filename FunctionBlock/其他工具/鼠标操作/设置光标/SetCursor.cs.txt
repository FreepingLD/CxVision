    {
        private userWcsRectangle1 _wcsRect1;
        public SetCursorParam Param { get; set; }

        [DisplayName("窗体坐标")]
        [DescriptionAttribute("输出属性")]
        public userWcsRectangle1 WcsRect1 { get => _wcsRect1; set => _wcsRect1 = value; }

        public SetCursor()
        {
            this.Param = new SetCursorParam();
            this.ResultInfo = new BindingList<MeasureResultInfo>();
            ((BindingList<MeasureResultInfo>)this.ResultInfo).Add(new MeasureResultInfo());
            ((BindingList<MeasureResultInfo>)this.ResultInfo).Add(new MeasureResultInfo());
            ((BindingList<MeasureResultInfo>)this.ResultInfo).Add(new MeasureResultInfo());
        }


        #region 实现接口

        public OperateResult Execute(params object[] param)
        {
            this.Result.Succss = true;
            this.Result.ExcuteState = enExcuteState.NONE;
            this.Result.ErrorMessage = "";
            string index = "";
            try
            {
                Stopwatch stopwatch = new Stopwatch();
                stopwatch.Start();
                ////////////////// 检测传入的参数是否包含有图像 ///////////////////////
                if (param != null)
                {
                    foreach (var item in param)
                    {
                        if (item != null)
                        {
                            switch (item.GetType().Name)
                            {
                                case nameof(Int32):
                                    int result = 0;
                                    if (int.TryParse(item.ToString(), out result))
                                        index = result.ToString();
                                    break;
                                case nameof(String):
                                    if (item.ToString().Split('=').Last().Length > 0)
                                    {
                                        if (int.TryParse(item.ToString().Split('=').Last(), out result))
                                            index = result.ToString();
                                    }
                                    break;
                            }
                        }
                    }
                }
                /////////////////////////////////////////
                this._wcsRect1 = new userWcsRectangle1();
                IntPtr winPtr = API.FindWindow(null, this.Param.WinName); //  "21CY TEST TOOL Ver 1.0.0"
                if (winPtr != IntPtr.Zero)
                {
                    RECT_INFO rect2 = new RECT_INFO();
                    API.GetWindowRect(winPtr, ref rect2); // 获取窗体的左上角
                    API.SetCursorPos(rect2.Left + this.Param.X, rect2.Top + this.Param.Y);
                    API.mouse_event(MouseEventFlag.LeftDown, rect2.Left + this.Param.X, rect2.Top + this.Param.Y, 1, UIntPtr.Zero);
                    API.mouse_event(MouseEventFlag.LeftUp, rect2.Left + this.Param.X, rect2.Top + this.Param.Y, 1, UIntPtr.Zero);
                    this._wcsRect1.X1 = rect2.Left;
                    this._wcsRect1.Y1 = rect2.Top;
                    this._wcsRect1.X2 = rect2.Right;
                    this._wcsRect1.Y2 = rect2.Bottom;
                    this.Result.Succss = true;
                }
                else
                    this.Result.Succss = false;
                stopwatch.Stop();
                ///////////////////////////////////////////
                this.CreateResultInfo(5);
                ((BindingList<MeasureResultInfo>)this.ResultInfo)[0].SetValue(this.name, "光标坐标_X", this._wcsRect1.X1 + this.Param.X);
                ((BindingList<MeasureResultInfo>)this.ResultInfo)[1].SetValue(this.name, "光标坐标_Y", this._wcsRect1.Y1 + this.Param.Y);
                ((BindingList<MeasureResultInfo>)this.ResultInfo)[2].SetValue(this.name, "窗口左上角_X", this._wcsRect1.X1);
                ((BindingList<MeasureResultInfo>)this.ResultInfo)[3].SetValue(this.name, "窗口左上角_Y", this._wcsRect1.Y1);
                ((BindingList<MeasureResultInfo>)this.ResultInfo)[4].SetValue(this.name, "时间(ms)", stopwatch.ElapsedMilliseconds);
            }
            catch (Exception ex)
            {
                LoggerHelper.Error(this.name + "->执行错误", ex);
                this.Result.Succss = false;
            }
            if (this.Result.Succss)
                LoggerHelper.Info(this.name + "->执行成功：" + $"光标坐标:{this._wcsRect1.X1 + this.Param.X},{this._wcsRect1.Y1 + this.Param.Y}");
            else
                LoggerHelper.Error(this.name + "->执行失败：" + $"光标坐标:{this._wcsRect1.X1 + this.Param.X},{this._wcsRect1.Y1 + this.Param.Y}");
            // 更改UI字体　
            UpdataNodeElementStyle(param, this.Result.Succss);
            return this.Result;
        }
        public object GetPropertyValues(string propertyName)
        {
            switch (propertyName)

                case nameof(this.Name):
                    return this.name;

            {

                default:
                    return true;
            }
        }

            }
        }
