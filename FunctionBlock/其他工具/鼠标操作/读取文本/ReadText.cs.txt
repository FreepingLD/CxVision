        public ReadText()
        {
            this.Param = new ReadTextParam();
            this.ResultInfo = new BindingList<OcrResultInfo>();
            ((BindingList<OcrResultInfo>)this.ResultInfo).Add(new OcrResultInfo());
            ((BindingList<OcrResultInfo>)this.ResultInfo).Add(new OcrResultInfo());

        #region 实现接口

        public OperateResult Execute(params object[] param)
        {
            this.Result.Succss = true;
            this.Result.ExcuteState = enExcuteState.NONE;
            this.Result.ErrorMessage = "";
            string index = "";
            try
            {
                Stopwatch stopwatch = new Stopwatch();
                stopwatch.Start();
                ////////////////// 检测传入的参数是否包含有图像 ///////////////////////
                if (param != null)
                {
                    foreach (var item in param)
                    {
                        if (item != null)
                        {
                            switch (item.GetType().Name)
                            {
                                case nameof(Int32):
                                    int result = 0;
                                    if (int.TryParse(item.ToString(), out result))
                                        index = result.ToString();
                                    break;
                                case nameof(String):
                                    if (item.ToString().Split('=').Last().Length > 0)
                                    {
                                        if (int.TryParse(item.ToString().Split('=').Last(), out result))
                                            index = result.ToString();
                                    }
                                    break;
                            }
                        }
                    }
                }
                /////////////////////////////////////////
                this._readContent = "";
                RECT_INFO rect2 = new RECT_INFO();
                IntPtr winPtr = API.FindWindow(null, this.Param.WinName); // 
                if (winPtr != IntPtr.Zero)
                {
                    API.GetWindowRect(winPtr, ref rect2); // 获取窗体的左上角
                    StringBuilder builder = new StringBuilder(100);
                    IntPtr ptr = API.WindowFromPoint(new Point(rect2.Left + this.Param.X, rect2.Top + this.Param.Y));   //rect2.Left + this.Param.X, rect2.Top + this.Param.Y
                    //IntPtr ptr = API.WindowFromPoint(winPtr, new Point(rect2.Left + this.Param.X, rect2.Top + this.Param.Y));
                    int length = API.GetWindowText(ptr, builder, 100);
                    //string lParam;
                    //API.SendMessage(ptr, API.WM_GETTEXT, 0, lParam);
                    this._readContent = builder.ToString();
                    this.Result.Succss = true;
                }
                else
                    this.Result.Succss = false;
                stopwatch.Stop();
                ///////////////////////////////////////////
                this.CreateResultInfo(4);
                ((BindingList<OcrResultInfo>)this.ResultInfo)[0].SetValue(this.name, "读取文本", this._readContent);
                ((BindingList<OcrResultInfo>)this.ResultInfo)[1].SetValue(this.name, "读取坐标_X", (rect2.Left + this.Param.X).ToString());
                ((BindingList<OcrResultInfo>)this.ResultInfo)[2].SetValue(this.name, "读取坐标_Y", (rect2.Top + this.Param.Y).ToString());
                ((BindingList<OcrResultInfo>)this.ResultInfo)[3].SetValue(this.name, "时间(ms)", stopwatch.ElapsedMilliseconds.ToString());
            }
            catch (Exception ex)
            {
                LoggerHelper.Error(this.name + "->执行错误", ex);
                this.Result.Succss = false;
            }
            if (this.Result.Succss)
                LoggerHelper.Info(this.name + "->执行成功：" + $"读取文本:{this._readContent}");
            else
                LoggerHelper.Error(this.name + "->执行失败：" + $"读取文本:{this._readContent}");
            // 更改UI字体　
            UpdataNodeElementStyle(param, this.Result.Succss);
            return this.Result;
        }
