        #region 实现接口

        public OperateResult Execute(params object[] param)
        {
            this.Result.Succss = true;
            this.Result.ExcuteState = enExcuteState.NONE;
            this.Result.ErrorMessage = "";
            string index = "";
            try
            {
                Stopwatch stopwatch = new Stopwatch();
                stopwatch.Start();
                ////////////////// 检测传入的参数是否包含有图像 ///////////////////////
                if (param != null)
                {
                    foreach (var item in param)
                    {
                        if (item != null)
                        {
                            switch (item.GetType().Name)
                            {
                                case nameof(Int32):
                                    int result = 0;
                                    if (int.TryParse(item.ToString(), out result))
                                        index = result.ToString();
                                    break;
                                case nameof(String):
                                    if (item.ToString().Split('=').Last().Length > 0)
                                    {
                                        if (int.TryParse(item.ToString().Split('=').Last(), out result))
                                            index = result.ToString();
                                    }
                                    break;
                            }
                        }
                    }
                }
                /////////////////////////////////////////
                IntPtr winPtr = API.FindWindow(null, this.Param.WinName); //  "21CY TEST TOOL Ver 1.0.0"
                if (winPtr != IntPtr.Zero)
                {
                    //RECT_INFO rect2 = new RECT_INFO();
                    //API.GetWindowRect(winPtr, ref rect2); // 获取窗体的左上角
                    //API.SetCursorPos(rect2.Left + 20, rect2.Top + 10);
                    //API.mouse_event(MouseEventFlag.LeftDown, rect2.Left + 20, rect2.Top + 10, 1, UIntPtr.Zero);
                    //API.mouse_event(MouseEventFlag.LeftUp, rect2.Left + 20, rect2.Top + 0, 1, UIntPtr.Zero);
                    API.SetForegroundWindow(winPtr);
                    switch (this.Param.KeyBoard)
                    {
                        default:
                        case enKeyBoard.Enter:
                            API.keybd_event(API.vbKeyEnter, 0, 0, 0);
                            break;
                        case enKeyBoard.Shift:
                            API.keybd_event(API.vbKeyShift, 0, 0, 0);
                            break;
                    }
                    this.Result.Succss = true;
                }
                else
                    this.Result.Succss = false;
                stopwatch.Stop();
                ///////////////////////////////////////////
                this.CreateResultInfo(2);
                ((BindingList<OcrResultInfo>)this.ResultInfo)[0].SetValue(this.name, "点击键", nameof(this.Param.KeyBoard));
                ((BindingList<OcrResultInfo>)this.ResultInfo)[1].SetValue(this.name, "时间(ms)", stopwatch.ElapsedMilliseconds.ToString());
            }
            catch (Exception ex)
            {
                LoggerHelper.Error(this.name + "->执行错误", ex);
                this.Result.Succss = false;
            }
            if (this.Result.Succss)
                LoggerHelper.Info(this.name + "->执行成功：" + $"点击键:{this.Param.KeyBoard}");
            else
                LoggerHelper.Error(this.name + "->执行失败：" + $"点击键:{this.Param.KeyBoard}");
            // 更改UI字体　
            UpdataNodeElementStyle(param, this.Result.Succss);
            return this.Result;
        }

            }
        }
