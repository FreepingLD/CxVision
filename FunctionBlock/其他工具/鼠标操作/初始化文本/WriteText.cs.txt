    [Serializable]

    public class WriteText : BaseFunction, IFunction
    {
        private string _writeContent;
        public WriteTextParam Param { get; set; }

        private string[] _content;

        [DisplayName("写入文本")]
        [DescriptionAttribute("输出属性")]
        public string WriteContent { get => _writeContent; set => _writeContent = value; }

        [DisplayName("输入文本")]
        [DescriptionAttribute("输入属性1")]
        public string[] Content
        {
            get
            {
                
                if (this.RefSource1.Count > 0)
                {
                    List<string> list = new List<string>();
                    object[] oo = this.GetPropertyValue(this.RefSource1);
                    if (oo != null && oo.Length > 0)
                    {
                        foreach (var item in oo)
                        {
                            switch (item.GetType().Name)
                            {
                                case nameof(String):
                                    string str = item as string;
                                    list.Add(str);
                                    break;
                                case "string[]":
                                    string[] strs = item as string[];
                                    if (strs != null)
                                    {
                                        foreach (var item2 in strs)
                                            list.Add(item2);
                                    }
                                    break;
                                default:
                                    throw new ArgumentException("不受支持的输入类型!");
                            }
                        }
                    }
                    this._content = list.ToArray();
                }
                else
                    this._content = new string[0];
                return this._content;
            }
            set
            {
                this._content = value;
            }
        }

        public WriteText()
        {
            this.Param = new WriteTextParam();
            this.ResultInfo = new BindingList<OcrResultInfo>();
            ((BindingList<OcrResultInfo>)this.ResultInfo).Add(new OcrResultInfo());
            ((BindingList<OcrResultInfo>)this.ResultInfo).Add(new OcrResultInfo());
        }


        #region 实现接口

        public OperateResult Execute(params object[] param)
        {
            this.Result.Succss = true;
            this.Result.ExcuteState = enExcuteState.NONE;
            this.Result.ErrorMessage = "";
            string index = "";
            try
            {
                Stopwatch stopwatch = new Stopwatch();
                stopwatch.Start();
                ////////////////// 检测传入的参数是否包含有图像 ///////////////////////
                if (param != null)
                {
                    foreach (var item in param)
                    {
                        if (item != null)
                        {
                            switch (item.GetType().Name)
                            {
                                case nameof(Int32):
                                    int result = 0;
                                    if (int.TryParse(item.ToString(), out result))
                                        index = result.ToString();
                                    break;
                                case nameof(String):
                                    if (item.ToString().Split('=').Last().Length > 0)
                                    {
                                        if (int.TryParse(item.ToString().Split('=').Last(), out result))
                                            index = result.ToString();
                                    }
                                    break;
                            }
                        }
                    }
                }
                /////////////////////////////////////////
                this._writeContent = "";
                RECT_INFO rect2 = new RECT_INFO();
                IntPtr winPtr = API.FindWindow(null, this.Param.WinName); //  "21CY TEST TOOL Ver 1.0.0"
                if (winPtr != IntPtr.Zero)
                {
                    API.GetWindowRect(winPtr, ref rect2); // 获取窗体的左上角
                    StringBuilder builder = new StringBuilder(100);
                    IntPtr ptr = API.WindowFromPoint(new Point(rect2.Left + this.Param.X, rect2.Top + this.Param.Y));
                    if (this.Content != null && this._content.Length > 0)
                    {
                        foreach (var item in this._content)
                            builder.Append(item);
                    }
                    else
                    {
                        builder.Append("send test");
                    }
                    //API.SetWindowText(ptr, builder);
                    API.SendMessage(ptr, API.WM_SETTEXT, 0,builder.ToString());
                    this._writeContent = builder.ToString();
                    this.Result.Succss = true;
                }
                else
                    this.Result.Succss = false;
                stopwatch.Stop();
                ///////////////////////////////////////////
                this.CreateResultInfo(4);
                ((BindingList<OcrResultInfo>)this.ResultInfo)[0].SetValue(this.name, "写入文本", this._writeContent);
                ((BindingList<OcrResultInfo>)this.ResultInfo)[1].SetValue(this.name, "写入坐标_X", (rect2.Left + this.Param.X).ToString());
                ((BindingList<OcrResultInfo>)this.ResultInfo)[2].SetValue(this.name, "写入坐标_Y", (rect2.Top + this.Param.Y).ToString());
                ((BindingList<OcrResultInfo>)this.ResultInfo)[3].SetValue(this.name, "时间(ms)", stopwatch.ElapsedMilliseconds.ToString());
            }
            catch (Exception ex)
            {
                LoggerHelper.Error(this.name + "->执行错误", ex);
                this.Result.Succss = false;
            }
            if (this.Result.Succss)
                LoggerHelper.Info(this.name + "->执行成功：" + $"写入文本:{this._writeContent}");
            else
                LoggerHelper.Error(this.name + "->执行失败：" + $"写入文本:{this._writeContent}");
            // 更改UI字体　
            UpdataNodeElementStyle(param, this.Result.Succss);
            return this.Result;
        }
        public object GetPropertyValues(string propertyName)
        {
            switch (propertyName)
            {
                default:

                    return this.name;
